cmake_minimum_required(VERSION 3.16)
project(rvizweb_native VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROS_VERSION "2" CACHE STRING "ROS adapter to compile: 1 or 2")
option(RVIZWEB_WITH_ROS "Build the native service (OFF builds protocol tests only)" ON)
find_package(Threads REQUIRED)
find_package(Boost 1.71 REQUIRED COMPONENTS system)
find_package(jsoncpp CONFIG REQUIRED)
add_library(rvizweb_protocol src/protocol.cpp)
target_include_directories(rvizweb_protocol PUBLIC include)
target_link_libraries(rvizweb_protocol PUBLIC JsonCpp::JsonCpp Threads::Threads)
target_compile_options(rvizweb_protocol PRIVATE -Wall -Wextra -Wpedantic)
if(RVIZWEB_WITH_ROS)
  add_library(rvizweb_ros_adapter STATIC)
  target_link_libraries(rvizweb_ros_adapter PUBLIC rvizweb_protocol)
  target_compile_definitions(rvizweb_ros_adapter PUBLIC RVIZWEB_ROS_VERSION=${ROS_VERSION})
  add_executable(rvizweb_native src/main.cpp src/server.cpp)
  target_link_libraries(rvizweb_native PRIVATE rvizweb_ros_adapter Boost::system)
  target_compile_options(rvizweb_native PRIVATE -Wall -Wextra -Wpedantic)
  if(ROS_VERSION STREQUAL "2")
    find_package(ament_cmake REQUIRED)
    find_package(rclcpp REQUIRED)
    find_package(sensor_msgs REQUIRED)
    find_package(nav_msgs REQUIRED)
    find_package(geometry_msgs REQUIRED)
    find_package(tf2_msgs REQUIRED)
    find_package(visualization_msgs REQUIRED)
    find_package(rosidl_typesupport_introspection_cpp REQUIRED)
    target_sources(rvizweb_ros_adapter PRIVATE src/ros2_adapter.cpp)
    ament_target_dependencies(rvizweb_ros_adapter PUBLIC rclcpp sensor_msgs geometry_msgs nav_msgs tf2_msgs visualization_msgs rosidl_typesupport_introspection_cpp)
  elseif(ROS_VERSION STREQUAL "1")
    # ROS1 headers/libraries also work with Ubuntu's native ROS packages.
    # CMAKE_PREFIX_PATH may point at /opt/ros/noetic or an isolated dependency prefix.
    find_path(ROS1_INCLUDE_DIR ros/ros.h REQUIRED)
    find_path(ROS1_MESSAGES_INCLUDE_DIR sensor_msgs/PointCloud2.h REQUIRED)
    target_sources(rvizweb_ros_adapter PRIVATE src/ros1_adapter.cpp)
    target_include_directories(rvizweb_ros_adapter PUBLIC ${ROS1_INCLUDE_DIR} ${ROS1_MESSAGES_INCLUDE_DIR})
    foreach(library roscpp rosconsole rostime roscpp_serialization topic_tools)
      find_library(ROS1_${library}_LIBRARY NAMES ${library} REQUIRED)
      target_link_libraries(rvizweb_ros_adapter PUBLIC ${ROS1_${library}_LIBRARY})
    endforeach()
  else()
    message(FATAL_ERROR "ROS_VERSION must be 1 or 2")
  endif()
  install(TARGETS rvizweb_native RUNTIME DESTINATION bin)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol_test tests/protocol_test.cpp)
  target_link_libraries(protocol_test PRIVATE rvizweb_protocol)
  add_test(NAME protocol COMMAND protocol_test)
  add_executable(display_test tests/display_test.cpp)
  target_link_libraries(display_test PRIVATE rvizweb_protocol)
  add_test(NAME display COMMAND display_test)
  add_executable(control_test tests/control_test.cpp)
  target_link_libraries(control_test PRIVATE rvizweb_protocol)
  add_test(NAME control COMMAND control_test)
  if(RVIZWEB_WITH_ROS)
    add_executable(control_fixture tests/control_fixture.cpp)
    target_link_libraries(control_fixture PRIVATE rvizweb_ros_adapter)
    add_executable(display_fixture tests/display_fixture.cpp)
    target_link_libraries(display_fixture PRIVATE rvizweb_ros_adapter)
    add_executable(ros_fixture tests/ros_fixture.cpp)
    target_link_libraries(ros_fixture PRIVATE rvizweb_ros_adapter)
  endif()
endif()
